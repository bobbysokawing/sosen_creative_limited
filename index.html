        const SPREADSHEET_ID = '159XP5OjDie2kuOYhbVyNXLk-CofaH8InNrRPzFPZQ4g';

        let currentLang = 'zh';

        let portfolioRawData = [];


        window.addEventListener('DOMContentLoaded', () => {

            loadGoogleSheetsData();

        async function loadGoogleSheetsData() {

            try {

                const urlText = `https://docs.google.com/spreadsheets/d/${SPREADSHEET_ID}/gviz/tq?tqx=out:csv&sheet=Text`;

                const urlPortfolio = `https://docs.google.com/spreadsheets/d/${SPREADSHEET_ID}/gviz/tq?tqx=out:csv&sheet=Portfolio`;


                const [resText, resPortfolio] = await Promise.all([

                    fetch(urlText).then(r => r.text()),

                    fetch(urlPortfolio).then(r => r.text())

                ]);


                parseAndApplyText(resText);

                parseAndApplyPortfolio(resPortfolio);


                document.getElementById('loader').style.opacity = '0';

                setTimeout(() => document.getElementById('loader').style.display = 'none', 500);

            } catch (error) {

                console.error("資料載入失敗:", error);

                document.getElementById('loader').innerText = "ERROR LOADING DATA. PLEASE CHECK SHEET.";

            }

        }


        function parseCSV(csvText) {

            const lines = csvText.split('\n');

            return lines.slice(1).map(line => {

                const matches = line.match(/(".*?"|[^",\s]+)(?=\s*,|\s*$)/g) || [];

                return matches.map(val => val.replace(/^"|"$/g, '').trim());

            }).filter(row => row.length > 0);

        }


        function parseAndApplyText(csvData) {

            const rows = parseCSV(csvData);

            rows.forEach(row => {

                const [id, zh, en] = row;

                

                if (id === 'logo_url') {

                    const txtNode = document.getElementById('logo-text-node');

                    const imgNode = document.getElementById('logo-img-node');

                    if (zh && (zh.startsWith('http://') || zh.startsWith('https://'))) {

                        txtNode.style.display = 'none';

                        imgNode.src = zh;

                        imgNode.style.display = 'block';

                    } else if (zh) {

                        imgNode.style.display = 'none';

                        txtNode.innerText = zh;

                        txtNode.style.display = 'block';

                    }

                    return;

                }


                const element = document.getElementById(`text-${id}`);

                if (element) {

                    if (id.startsWith('dim_product') || id.startsWith('dim_promotion') || id.startsWith('dim_brand')) {

                        const listZh = zh.split(/[；，分;]/).map(item => item.trim()).filter(i => i).map(i => `<li>${i}</li>`).join('');

                        const listEn = en.split(/[；，分;]/).map(item => item.trim()).filter(i => i).map(i => `<li>${i}</li>`).join('');

                        element.innerHTML = `<ul class="lang-zh">${listZh}</ul><ul class="lang-en">${listEn}</ul>`;

                    } else {

                        element.innerHTML = `<span class="lang-zh">${zh}</span><span class="lang-en">${en}</span>`;

                    }

                }

            });

        }


        function parseAndApplyPortfolio(csvData) {

            const rows = parseCSV(csvData);

            portfolioRawData = rows;

            const grid = document.getElementById('dynamic-portfolio-grid');

            grid.innerHTML = ''; 


            rows.forEach((row, index) => {

                const [type, image, titleZh, titleEn, descZh, descEn, heroImage, detailZh, detailEn] = row;


                if (type === 'real') {

                    const item = document.createElement('div');

                    item.className = 'portfolio-item';

                    item.onclick = () => openProjectDetail(index);

                    item.innerHTML = `

                        <div class="image-wrapper"><img class="portfolio-img" src="${image}" alt="Portfolio"></div>

                        <div class="item-info">

                            <h3><span class="lang-zh">${titleZh}</span><span class="lang-en">${titleEn}</span></h3>

                            <p><span class="lang-zh">${descZh}</span><span class="lang-en">${descEn}</span></p>

                        </div>

                    `;

                    grid.appendChild(item);

                } else if (type.startsWith('donut')) {

                    const section = document.getElementById(`donut-case-${type}`);

                    if (section) {

                        section.querySelector('.target-title').innerHTML = `<span class="lang-zh">${titleZh}</span><span class="lang-en">${titleEn}</span>`;

                        section.querySelector('.target-desc').innerHTML = `<span class="lang-zh">${descZh}</span><span class="lang-en">${descEn}</span>`;

                        section.querySelector('.target-img').src = image;

                    }

                }

            });

        }


        function openProjectDetail(index) {

            const row = portfolioRawData[index];

            const [type, image, titleZh, titleEn, descZh, descEn, heroImage, detailZh, detailEn] = row;


            document.getElementById('detail-hero-img').src = heroImage || image;

            document.getElementById('detail-title').innerHTML = `<span class="lang-zh">${titleZh}</span><span class="lang-en">${titleEn}</span>`;

            document.getElementById('detail-tags').innerHTML = `<span class="lang-zh">${descZh}</span><span class="lang-en">${descEn}</span>`;

            document.getElementById('detail-text').innerHTML = `<span class="lang-zh">${detailZh}</span><span class="lang-en">${detailEn}</span>`;


            switchPage('project-detail');

        }


        function toggleLanguage() {

            const body = document.body;

            const btn = document.querySelector('.lang-switch');

            if (currentLang === 'zh') {

                body.classList.add('lang-en');

                btn.innerText = 'ZH';

                currentLang = 'en';

            } else {

                body.classList.remove('lang-en');

                btn.innerText = 'EN';

                currentLang = 'zh';

            }

        }


        function switchPage(pageId) {

            document.querySelectorAll('.page').forEach(page => page.classList.remove('active'));

            document.querySelectorAll('.nav-links span').forEach(link => link.classList.remove('active'));

            

            const targetPage = document.getElementById(pageId);

            if(targetPage) targetPage.classList.add('active');

            

            const targetLink = document.getElementById('link-' + pageId);

            if (targetLink) targetLink.classList.add('active');

            

            window.scrollTo({ top: 0, behavior: 'smooth' });

        }
